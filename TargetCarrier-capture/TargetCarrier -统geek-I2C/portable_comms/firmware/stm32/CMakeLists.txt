cmake_minimum_required(VERSION 3.22)
project(tc_stm32 C ASM)
set(CMAKE_C_STANDARD 11)
set(TC_BACKEND 3 CACHE STRING "1 UART, 2 SPI, 3 I2C")
set(TC_STM_BASE "${CMAKE_CURRENT_LIST_DIR}/../../reference/stm32" CACHE PATH "Reference Cube project (in working copy)")
set(ROOT ${CMAKE_CURRENT_LIST_DIR}/../..)
add_executable(tc_stm32 main.c ${ROOT}/core/tc_protocol.c ${ROOT}/core/tc_params.c ${ROOT}/core/tc_wave.c ${ROOT}/platform/stm32_bus.c
 ${TC_STM_BASE}/startup_stm32f103xb.s ${TC_STM_BASE}/Core/Src/system_stm32f1xx.c)
target_include_directories(tc_stm32 PRIVATE ${ROOT}/core ${ROOT}/config ${ROOT}/platform ${TC_STM_BASE}/Drivers/CMSIS/Include ${TC_STM_BASE}/Drivers/CMSIS/Device/ST/STM32F1xx/Include)
target_compile_definitions(tc_stm32 PRIVATE STM32F103xB HSE_VALUE=8000000 TC_BACKEND=${TC_BACKEND})
target_compile_options(tc_stm32 PRIVATE -mcpu=cortex-m3 -mthumb -Os -ffunction-sections -fdata-sections -Wall -Wextra -Wno-unused-variable -Wno-unused-function)
target_link_options(tc_stm32 PRIVATE -mcpu=cortex-m3 -mthumb -T${TC_STM_BASE}/STM32F103XX_FLASH.ld -Wl,--gc-sections -Wl,-Map=tc_stm32.map --specs=nano.specs --specs=nosys.specs)
target_link_libraries(tc_stm32 m)
add_custom_command(TARGET tc_stm32 POST_BUILD COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:tc_stm32> tc_stm32.bin COMMAND ${CMAKE_SIZE} $<TARGET_FILE:tc_stm32>)
